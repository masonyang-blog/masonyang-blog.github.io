1.1 核心計算公式與造市商假設

[Direct Answer] 市場流動性的微觀摩擦力，往往源自造市商在衍生品帳簿上的機械性對沖。要量化整體市場的 Gamma Exposure（GEX），業界通用 SpotGamma 與 SqueezeMetrics 的標準造市商假設基準：在 Call（買權） 端，假設散戶與機構傾向賣出 Covered Call 賺取權利金，造市商被迫成為買方（Long Call），因此具備 正 Gamma（+1），價格上漲時造市商必須賣出期貨避險，形成天然阻力；在 Put（賣權） 端，假設機構傾向買進保護性賣權對沖尾部風險，造市商成為賣方（Short Put），因此具備 負 Gamma（-1），價格下跌時造市商被迫追空期貨以維持 Delta 中性，進而放大市場脆弱性。

[Support Details] 單一履約價的 GEX 計算公式，定義為標的指數每變動 1% 時造市商所承受的美元避險曝險：

$$GEX_{\text{strike}} = \text{Gamma} \times OI \times 100 \times S^2 \times 0.01$$

（其中 $S$ 為標的現價，$100$ 為每口美股選擇權乘數）

import datetime as dt
                    import matplotlib.pyplot as plt
                    import numpy as np
                    import pandas as pd
                    from scipy.stats import norm
                    import yfinance as yf


                    # ---------------------------------------------------------
                    # 1. Black-Scholes Gamma 計算函式
                    # ---------------------------------------------------------
                    def calc_bs_gamma(S, K, T, r, sigma):
                        """計算歐式選擇權的 Black-Scholes Gamma (Call 與 Put 的 Gamma 理論值相同)

                        S: 標的現價 | K: 履約價 | T: 距到期年化時間 | r: 無風險利率 | sigma:
                        隱含波動率(IV)
                        """
                        # 避免到期時間為 0 或 IV 異常導致除以零
                        T = np.maximum(T, 1e-5)
                        sigma = np.maximum(sigma, 1e-4)

                        d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
                        gamma = norm.pdf(d1) / (S * sigma * np.sqrt(T))
                        return gamma


                    # ---------------------------------------------------------
                    # 2. 抓取 SPY 選擇權鏈並計算 GEX
                    # ---------------------------------------------------------
                    def get_spy_gex_profile(ticker_symbol="SPY", risk_free_rate=0.042, max_dte=45):
                        ticker = yf.Ticker(ticker_symbol)

                        # 取得最新收盤價 / 現價
                        hist = ticker.history(period="1d")
                        spot_price = hist["Close"].iloc[-1]
                        print(f"[{ticker_symbol}] 目前現價: ${spot_price:.2f}")

                        today = dt.datetime.now().date()
                        expirations = ticker.options

                        all_options = []

                        for exp_str in expirations:
                            exp_date = dt.datetime.strptime(exp_str, "%Y-%m-%d").date()
                            dte = (exp_date - today).days

                            # 過濾掉已到期或超過 max_dte 天的遠月合約（Gamma 主要集中在近 45 天內）
                            if dte < 0 or dte > max_dte:
                                continue

                            # 轉換為年化時間 T（若為 0DTE 當天到期，給予半個交易日的殘值避免除以零）
                            T = max(dte, 0.5) / 365.0

                            opt_chain = ticker.option_chain(exp_str)

                            # 處理 Calls
                            calls = opt_chain.calls[["strike", "openInterest", "impliedVolatility"]].copy()
                            calls["type"] = "call"
                            calls["T"] = T

                            # 處理 Puts
                            puts = opt_chain.puts[["strike", "openInterest", "impliedVolatility"]].copy()
                            puts["type"] = "put"
                            puts["T"] = T

                            all_options.append(calls)
                            all_options.append(puts)

                        df = pd.concat(all_options, ignore_index=True)

                        # 清理無效數據（剔除 OI 為 0 或 IV 異常的合約）
                        df = df.dropna(subset=["openInterest", "impliedVolatility"])
                        df = df[(df["openInterest"] > 0) & (df["impliedVolatility"] > 0.01)]

                        # 計算每個合約的 Gamma
                        df["gamma"] = calc_bs_gamma(
                            S=spot_price,
                            K=df["strike"].values,
                            T=df["T"].values,
                            r=risk_free_rate,
                            sigma=df["impliedVolatility"].values,
                        )

                        # 計算 GEX（單位：十億美元 $ Billions / per 1% move）
                        # Call 為正 Gamma (+1)，Put 為負 Gamma (-1)
                        sign = np.where(df["type"] == "call", 1.0, -1.0)
                        df["gex"] = (
                            sign * df["gamma"] * df["openInterest"] * 100 * (spot_price**2) * 0.01
                        ) / 1e9

                        # 依履約價加總
                        gex_by_strike = df.groupby("strike").agg(
                            net_gex=("gex", "sum"),
                            call_gex=("gex", lambda x: x[x > 0].sum()),
                            put_gex=("gex", lambda x: x[x < 0].sum()),
                        ).reset_index()

                        # 只保留現價上下 ±12% 範圍內的履約價以便視覺化與精準計算
                        lower_bound = spot_price * 0.88
                        upper_bound = spot_price * 1.12
                        gex_filtered = gex_by_strike[
                            (gex_by_strike["strike"] >= lower_bound)
                            & (gex_by_strike["strike"] <= upper_bound)
                        ].copy()

                        return spot_price, gex_filtered


                    # ---------------------------------------------------------
                    # 3. 尋找 Call Wall、Put Wall 與 Gamma Flip 翻轉點
                    # ---------------------------------------------------------
                    def analyze_levels(spot_price, gex_df):
                        # Call Wall: 正 GEX 最大的履約價
                        call_wall_row = gex_df.loc[gex_df["call_gex"].idxmax()]
                        call_wall = call_wall_row["strike"]

                        # Put Wall: 負 GEX 最深（最小值）的履約價
                        put_wall_row = gex_df.loc[gex_df["put_gex"].idxmin()]
                        put_wall = put_wall_row["strike"]

                        # Gamma Flip: 尋找淨 GEX 從負轉正的交叉點（利用線性內插法求精確價位）
                        gex_df = gex_df.sort_values("strike").reset_index(drop=True)
                        # 計算累積或局部移動平均以消除單一履約價的雜訊跳動
                        gex_df["smooth_gex"] = gex_df["net_gex"].rolling(window=3, min_periods=1, center=True).mean()

                        gamma_flip = None
                        # 從現價附近往上下搜尋符號反轉點
                        for i in range(len(gex_df) - 1):
                            g1 = gex_df.loc[i, "smooth_gex"]
                            g2 = gex_df.loc[i + 1, "smooth_gex"]
                            k1 = gex_df.loc[i, "strike"]
                            k2 = gex_df.loc[i + 1, "strike"]

                            if g1 < 0 and g2 >= 0:
                                # 線性內插計算零點
                                gamma_flip = k1 + (0 - g1) * (k2 - k1) / (g2 - g1)
                                # 優先取最靠近現價的翻轉點
                                if abs(gamma_flip - spot_price) < spot_price * 0.08:
                                    break

                        total_net_gex = gex_df["net_gex"].sum()

                        print("\n--- 機構選擇權結構關鍵水位 ---")
                        print(f"整體淨 Gamma (Net GEX): {total_net_gex:+.2f} Billion ($/1% move)")
                        print(f"Call Wall (最大阻力/天花板): ${call_wall:.2f}")
                        print(f"Put Wall  (最大支撐/地板)  : ${put_wall:.2f}")
                        if gamma_flip:
                            print(f"Gamma Flip (正負翻轉警戒線): ${gamma_flip:.2f}")
                            regime = "正 Gamma (穩定器/高拋低吸)" if spot_price > gamma_flip else "負 Gamma (加速器/追漲殺跌)"
                            print(f"目前市場結構狀態           : {regime}")
                        else:
                            print("Gamma Flip: 當前區間內未出現正負反轉點（全區間單邊主導）")

                        return call_wall, put_wall, gamma_flip


                    # ---------------------------------------------------------
                    # 4. 繪製 GEX 結構分布圖（銳利幾何終端機風格）
                    # ---------------------------------------------------------
                    def plot_gex_profile(spot_price, gex_df, call_wall, put_wall, gamma_flip, ticker="SPY"):
                        plt.style.use("dark_background")
                        fig, ax = plt.subplots(figsize=(12, 6), dpi=120)

                        strikes = gex_df["strike"]
                        net_gex = gex_df["net_gex"]

                        # 正 GEX 用青綠色，負 GEX 用珊瑚紅
                        colors = ["#00e676" if val >= 0 else "#ff5252" for val in net_gex]

                        ax.bar(strikes, net_gex, width=0.8, color=colors, alpha=0.85, edgecolor="none")
                        ax.axhline(0, color="#888888", linewidth=1.0)

                        # 標示現價 Spot
                        ax.axvline(spot_price, color="#ffffff", linestyle="-", linewidth=1.8, label=f"Spot Price: ${spot_price:.2f}")

                        # 標示 Call Wall 與 Put Wall
                        ax.axvline(call_wall, color="#00e676", linestyle="--", linewidth=1.3, label=f"Call Wall: ${call_wall:.0f}")
                        ax.axvline(put_wall, color="#ff5252", linestyle="--", linewidth=1.3, label=f"Put Wall: ${put_wall:.0f}")

                        # 標示 Gamma Flip
                        if gamma_flip:
                            ax.axvline(gamma_flip, color="#ffd740", linestyle="-.", linewidth=1.8, label=f"Gamma Flip: ${gamma_flip:.2f}")

                        ax.set_title(f"{ticker} Net Gamma Exposure (GEX) Profile by Strike", fontsize=14, fontweight="bold", pad=14)
                        ax.set_xlabel("Strike Price ($)", fontsize=11)
                        ax.set_ylabel("Net GEX ($ Billions per 1% Move)", fontsize=11)
                        ax.grid(True, linestyle=":", alpha=0.25)
                        ax.legend(loc="upper left", frameon=True, facecolor="#1e1e1e", edgecolor="#444444")

                        plt.tight_layout()
                        plt.show()


                    if __name__ == "__main__":
                        spot, gex_data = get_spy_gex_profile(ticker_symbol="SPY", max_dte=45)
                        c_wall, p_wall, g_flip = analyze_levels(spot, gex_data)
                        plot_gex_profile(spot, gex_data, c_wall, p_wall, g_flip, ticker="SPY")

1.3 實戰使用與擴充微調建議

在實際部署這套原型時，參數邊界與資料清洗決定了訊號的信噪比。首先，程式預設將到期天數篩選上限設為 `max_dte=45`，原因在於 Black-Scholes 模型中的 Gamma 分母包含 $\sqrt{T}$，遠月合約（例如半年或一年後到期）的 Gamma 值極度微小，對造市商日內動態對沖的邊際影響幾乎可以忽略；相反地，0DTE 至 45 DTE 的近月合約集中了市場 80% 以上的 Gamma 流動性。若操作週期聚焦於當日或當週的短線結算擠壓效應，可進一步將參數收斂為 `max_dte=7`。

2.1 核心希臘字母與造市商對沖公式

Vanna（波動率敏感度：$\frac{\partial \Delta}{\partial \sigma}$） 衡量隱含波動率（IV）變動 1% 時，造市商避險期貨部位的變化量。其數學定義為：

$$\text{Vanna} = -\phi(d_1) \frac{d_2}{\sigma}$$

在進行造市商回補估算時，當市場恐慌緩解、波動率從高點回落 $\Delta\sigma$（例如 `-0.05`，即隱含波動率下降 5%）之際，造市商原本為了防禦下跌而建立的期貨空單會瞬間變成過度避險，被迫在期貨市場買回的契約口數為：

$$\Delta \text{Futures}_{\text{Vanna}} = -\sum (\text{Sign} \times \text{Vanna} \times OI \times \Delta\sigma)$$

Charm（時間衰減敏感度：$\frac{\partial \Delta}{\partial t}$） 則衡量時間流逝 $\Delta t$（例如距離收盤或結算僅剩最後 2 小時）時，選擇權 Delta 自動向 0 或 1 收斂歸零的速度，這也是結算日尾盤自動逼空的物理根源：

$$\text{Charm}_{\text{Call}} = -\phi(d_1) \frac{2rT - d_2 \sigma \sqrt{T}}{2T \sigma \sqrt{T}}, \quad \text{Charm}_{\text{Put}} = \text{Charm}_{\text{Call}} + r \cdot N(-d_1) \text{（忽略股息）}$$

註：實務上無股息歐式選擇權之 Put Charm 簡化為 $\text{Charm}_{\text{Call}}$ 加上折現修正項，兩者在近月極接近。

                    # 1. Black-Scholes 定價、IV 反推與高階 Greeks (Gamma / Vanna / Charm)

                    def bs_price(S, K, T, r, sigma, option_type="call"):
                        d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
                        d2 = d1 - sigma * np.sqrt(T)
                        if option_type == "call":
                            return S * norm.cdf(d1) - K * np.exp(-r * T) * norm.cdf(d2)
                        else:
                            return K * np.exp(-r * T) * norm.cdf(-d2) - S * norm.cdf(-d1)


                    def implied_volatility(price, S, K, T, r, option_type="call"):
                        """使用牛頓法 (Newton-Raphson) 從期交所收盤價反推真實 IV"""
                        if price <= 0 or T <= 0:
                            return np.nan
                        # 檢查是否低於內含價值（無套利下限）
                        intrinsic = max(0.0, S - K) if option_type == "call" else max(0.0, K - S)
                        if price <= intrinsic:
                            return 0.05  # 給予極低底線 IV 避免發散

                        sigma = 0.20  # 初始猜測 20%
                        for _ in range(50):
                            p = bs_price(S, K, T, r, sigma, option_type)
                            diff = p - price
                            if abs(diff) < 1e-4:
                                return sigma
                            d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
                            vega = S * norm.pdf(d1) * np.sqrt(T)
                            if vega < 1e-6:
                                break
                            sigma -= diff / vega
                            if sigma <= 0.005 or sigma > 3.0:
                                return np.nan
                        return sigma


                    def calc_greeks_vectorized(S, K, T, r, sigma, opt_types):
                        """向量化計算 Gamma, Vanna, Charm"""
                        T = np.maximum(T, 1e-5)
                        sigma = np.maximum(sigma, 1e-4)

                        d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
                        d2 = d1 - sigma * np.sqrt(T)
                        pdf_d1 = norm.pdf(d1)

                        # 1. Gamma (Call 與 Put 相同)
                        gamma = pdf_d1 / (S * sigma * np.sqrt(T))

                        # 2. Vanna (dDelta / dSigma)
                        vanna = -pdf_d1 * (d2 / sigma)

                        # 3. Charm (dDelta / dT，年化時間變化率)
                        charm_common = -pdf_d1 * (2 * r * T - d2 * sigma * np.sqrt(T)) / (2 * T * sigma * np.sqrt(T))
                        charm = np.where(opt_types == "call", charm_common, charm_common)

                        return gamma, vanna, charm

                                T=row.T,
                                r=risk_free_rate,
                                option_type=row.買賣權,
                            )
                            ivs.append(iv)
                        df_near["iv"] = ivs
                        df_near = df_near.dropna(subset=["iv"]).copy()

                        # 計算 Greeks
                        gamma, vanna, charm = calc_greeks_vectorized(
                            S=spot_index,
                            K=df_near["履約價"].values,
                            T=df_near["T"].values,
                            r=risk_free_rate,
                            sigma=df_near["iv"].values,
                            opt_types=df_near["買賣權"].values,
                        )

                        df_near["gamma"] = gamma
                        df_near["vanna"] = vanna
                        df_near["charm"] = charm

                        # 造市商部位方向假設：Call 為正 Gamma (+1)，Put 為負 Gamma (-1)
                        # 台指選乘數為每點 50 元台幣；小台指(MTX)乘數亦為每點 50 元（比例為 1:1）
                        sign = np.where(df_near["買賣權"] == "call", 1.0, -1.0)

                        # 1. GEX：大盤每變動 1% (spot * 0.01)，造市商必須對沖的「小台指(MTX)口數」
                        df_near["gex_mtx_lots"] = sign * df_near["gamma"] * df_near["未平倉契約數"] * (spot_index * 0.01)

                        # 2. Vanna 回補口數：當 IV 下降 iv_drop_sim (-5%) 時，造市商被迫買回的小台指口數
                        # 註：Short Put 在 IV 下降時 Delta 上升（空單過多），因此造市商需「買回」期貨
                        df_near["vanna_rebuy_mtx"] = -sign * df_near["vanna"] * df_near["未平倉契約數"] * iv_drop_sim

                        # 3. Charm 回補口數：當時間推進 hours_to_close 小時，造市商被迫買回的小台指口數
                        dt_years = -(hours_to_close / 24.0) / 365.0
                        df_near["charm_rebuy_mtx"] = -sign * df_near["charm"] * df_near["未平倉契約數"] * dt_years

                        # 依履約價彙整
                        summary_df = df_near.groupby("履約價").agg(
                            net_gex=("gex_mtx_lots", "sum"),
                            call_gex=("gex_mtx_lots", lambda x: x[x > 0].sum()),
                            put_gex=("gex_mtx_lots", lambda x: x[x < 0].sum()),
                            vanna_flow=("vanna_rebuy_mtx", "sum"),
                            charm_flow=("charm_rebuy_mtx", "sum"),
                        ).reset_index()

                        # 過濾現價上下 ±8% 範圍

                        ].sort_values("履約價").reset_index(drop=True)

                        # 找出三大關鍵水位
                        call_wall = summary_df.loc[summary_df["call_gex"].idxmax(), "履約價"]
                        put_wall = summary_df.loc[summary_df["put_gex"].idxmin(), "履約價"]

3.1 為什麼要加入「三大法人台指期未平倉」做雙重驗證？

[Support Details] 單看造市商的選擇權 Gamma 結構（GEX）存在一個結構性盲點：公開未平倉量無法直接區分大額 Put OI 究竟是外資主動買進避險，還是自營商單純賣出收租。將「外資台指期（TX）淨未平倉口數」與「造市商 Net GEX」進行交叉比對，才能剝開籌碼表象，精準辨識機構的真實意圖：

造市商結構 (TXO GEX) 外資台指期淨部位 (TX OI) 市場真實劇本與機構定價邏輯 操作建議
正 Gamma (在 Gamma Flip 之上) 外資淨多單增加或小幅空單 【健康慢牛 / 區間震盪】 外資現貨偏多，選擇權莊家提供下檔支撐，逢回皆有買盤承接。 沿 Put Wall 偏多操作，突破 Call Wall 不追高。
正 Gamma (在 Gamma Flip 之上) 外資淨空單異常大增 (如破 4 萬口) 【高檔鎖利避險】 外資手握大量現貨，同時在期貨布空單＋賣 Call（Covered Call）鎖住獲利，天花板極硬。 嚴禁追高，反彈至 Call Wall 附近適合分批減碼或做賣方。
負 Gamma (跌破 Gamma Flip) 外資淨空單持續擴大 【機構單邊狙擊 / 流動性雪崩】 外資期現貨同步偏空，加上造市商被迫追殺賣空期貨，下跌動能最強。 順勢偏空操作，絕對不可在盤中盲目摸底接刀。
負 Gamma (跌破 Gamma Flip) 外資淨空單大幅回補減少 【空頭力竭 / V 轉前兆】 外資開始平倉期貨空單，一旦大盤止跌引發 VIX 回落，造市商 Vanna/Charm 回補將觸發暴力軋空。 空單立刻獲利了結，準備迎接右側 V 型反轉買點。

                    # =========================================================
                    # 1. Black-Scholes 定價、IV 反推與高階 Greeks
                    # =========================================================
                    def bs_price(S, K, T, r, sigma, option_type="call"):
                        d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
                        d2 = d1 - sigma * np.sqrt(T)
                        if option_type == "call":
                            return S * norm.cdf(d1) - K * np.exp(-r * T) * norm.cdf(d2)
                        return K * np.exp(-r * T) * norm.cdf(-d2) - S * norm.cdf(-d1)


                    def implied_volatility(price, S, K, T, r, option_type="call"):
                        if price <= 0 or T <= 0:
                            return np.nan
                        intrinsic = max(0.0, S - K) if option_type == "call" else max(0.0, K - S)
                        if price <= intrinsic:
                            return 0.05

                        sigma = 0.20
                        for _ in range(50):
                            p = bs_price(S, K, T, r, sigma, option_type)
                            diff = p - price
                            if abs(diff) < 1e-4:
                                return sigma
                            d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
                            vega = S * norm.pdf(d1) * np.sqrt(T)
                            if vega < 1e-6:
                                break
                            sigma -= diff / vega
                            if sigma <= 0.005 or sigma > 3.0:
                                return np.nan
                        return sigma


                    def calc_greeks_vectorized(S, K, T, r, sigma, opt_types):
                        T = np.maximum(T, 1e-5)
                        sigma = np.maximum(sigma, 1e-4)
                        d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
                        d2 = d1 - sigma * np.sqrt(T)
                        pdf_d1 = norm.pdf(d1)

                        gamma = pdf_d1 / (S * sigma * np.sqrt(T))
                        vanna = -pdf_d1 * (d2 / sigma)
                        charm = -pdf_d1 * (2 * r * T - d2 * sigma * np.sqrt(T)) / (2 * T * sigma * np.sqrt(T))
                        return gamma, vanna, charm


                    # =========================================================
                    # 2. 精算台灣期交所週三結算日 DTE
                    # =========================================================
                    def estimate_txo_dte(contract_code, current_date):
                        """根據期交所合約代碼（如 '202611' 或 '202611W4'）推算剩餘天數"""
                        try:
                            ym_str = contract_code[:6]
                            year, month = int(ym_str[:4]), int(ym_str[4:6])
                            cal = calendar.monthcalendar(year, month)
                            # 找出該月所有的週三（weekday == 2）
                            wednesdays = [week[calendar.WEDNESDAY] for week in cal if week[calendar.WEDNESDAY] != 0]

                            if "W" in contract_code:
                                week_num = int(contract_code.split("W")[1])
                                settle_day = wednesdays[min(week_num - 1, len(wednesdays) - 1)]
                            else:
                                # 月選為第三個週三
                                settle_day = wednesdays[2]

                            settle_date = dt.date(year, month, settle_day)
                            dte = (settle_date - current_date).days
                            return max(float(dte), 0.5)  # 結算當天給予 0.5 天殘值避免除以零
                        except Exception:
                            return 3.0 if "W" in contract_code else 15.0

                    # 3. 抓取數據：加權指數、TXO 選擇權、三大法人台指期未平倉

                    def get_taiex_close(query_date_obj):
                        """從證交所公開 API 抓取當日加權指數收盤價"""

                                    return float(row[1].replace(",", ""))
                        raise ValueError("無法取得加權指數收盤價（可能為休市日或資料尚未更新）。")


                    def get_institutional_futures_oi(query_date_str):
                        """從期交所抓取三大法人「臺股期貨(TX)」淨未平倉口數"""
                        url = "https://www.taifex.com.tw/cht/3/futContractsDateDown"

                        url = "https://www.taifex.com.tw/cht/3/optDataDown"
                        payload = {
                            "down_type": "1",
                            "queryStartDate": query_date_str,
                            "queryEndDate": query_date_str,

                    # 4. 核心計算與雙重驗證判讀引擎
                    # =========================================================
                    def run_analysis():
                        # 使用台灣時區 (UTC+8)
                        tw_tz = dt.timezone(dt.timedelta(hours=8))
                        now_tw = dt.datetime.now(tw_tz).date()
                        query_date_str = now_tw.strftime("%Y/%m/%d")

                        # 1. 取得加權指數（若今日休市會直接拋出例外並安全結束排程）
                        spot_index = get_taiex_close(now_tw)
                        foreign_oi, trust_oi = get_institutional_futures_oi(query_date_str)
                        df = fetch_txo_csv(query_date_str)
                        df = df[(df["未平倉契約數"] > 0) & (df["收盤價"] > 0)].copy()

                        # 鎖定未平倉量最大的最近兩個合約
                        top_contracts = df.groupby("到期月份(週別)")["未平倉契約數"].sum().nlargest(2).index.tolist()
                        df_near = df[df["到期月份(週別)"].isin(top_contracts)].copy()

                        # 精算 DTE 與反推 IV
                        df_near["dte"] = df_near["到期月份(週別)"].apply(lambda c: estimate_txo_dte(c, now_tw))
                        df_near["T"] = df_near["dte"] / 365.0
                        r = 0.015

                        df_near["iv"] = [
                            implied_volatility(row.收盤價, spot_index, row.履約價, row.T, r, row.買賣權)
                            for row in df_near.itertuples()
                        ]
                        df_near = df_near.dropna(subset=["iv"]).copy()

                        gamma, vanna, charm = calc_greeks_vectorized(
                            spot_index, df_near["履約價"].values, df_near["T"].values, r, df_near["iv"].values, df_near["買賣權"].values
                        )
                        sign = np.where(df_near["買賣權"] == "call", 1.0, -1.0)

                        df_near["gex_mtx"] = sign * gamma * df_near["未平倉契約數"] * (spot_index * 0.01)
                        df_near["vanna_rebuy"] = -sign * vanna * df_near["未平倉契約數"] * (-0.05)  # IV 降 5%
                        df_near["charm_rebuy"] = -sign * charm * df_near["未平倉契約數"] * (-(2.0 / 24.0) / 365.0)  # 剩 2 小時

                        summary = df_near.groupby("履約價").agg(
                            net_gex=("gex_mtx", "sum"),
                            call_gex=("gex_mtx", lambda x: x[x > 0].sum()),
                            put_gex=("gex_mtx", lambda x: x[x < 0].sum()),
                            vanna_flow=("vanna_rebuy", "sum"),
                            charm_flow=("charm_rebuy", "sum"),
                        ).reset_index()

                        summary = summary[(summary["履約價"] >= spot_index * 0.92) & (summary["履約價"] <= spot_index * 1.08)].sort_values("履約價").reset_index(drop=True)

                        call_wall = float(summary.loc[summary["call_gex"].idxmax(), "履約價"])
                        put_wall = float(summary.loc[summary["put_gex"].idxmin(), "履約價"])

                        summary["smooth_gex"] = summary["net_gex"].rolling(3, min_periods=1, center=True).mean()
                        gamma_flip = None
                        for i in range(len(summary) - 1):
                            g1, g2 = summary.loc[i, "smooth_gex"], summary.loc[i + 1, "smooth_gex"]
                            k1, k2 = summary.loc[i, "履約價"], summary.loc[i + 1, "履約價"]
                            if g1 < 0 and g2 >= 0:
                                gamma_flip = round(float(k1 + (0 - g1) * (k2 - k1) / (g2 - g1)), 1)
                                if abs(gamma_flip - spot_index) < spot_index * 0.05:
                                    break

                        is_pos_gamma = True if (gamma_flip is None or spot_index > gamma_flip) else False
                        regime_str = "正 Gamma (高拋低吸)" if is_pos_gamma else "負 Gamma (追漲殺跌)"

                        # 雙重驗證戰略訊號
                        if is_pos_gamma and foreign_oi > -25000:
                            signal = "健康區間震盪：逢回 Put Wall 偏多，Call Wall 停利"
                        elif is_pos_gamma and foreign_oi <= -25000:
                            signal = "高檔避險鎖利：天花板極硬，反彈至 Call Wall 試空/減碼"
                        elif not is_pos_gamma and foreign_oi <= -30000:
                            signal = "負 Gamma 雙殺雪崩：外資重空+莊家追殺，順勢偏空勿摸底"
                        else:
                            signal = "負 Gamma 潛在逼空：密切關注 VIX 回落觸發 Vanna 急拉"

                        return {
                            "date": query_date_str,
                            "spot": spot_index,
                            "call_wall": call_wall,
                            "put_wall": put_wall,
                            "gamma_flip": gamma_flip or "N/A",
                            "regime": regime_str,
                            "net_gex_mtx": round(float(summary["net_gex"].sum()), 0),
                            "vanna_rebuy_mtx": round(float(summary["vanna_flow"].sum()), 0),
                            "charm_rebuy_mtx": round(float(summary["charm_flow"].sum()), 0),
                            "foreign_tx_oi": foreign_oi,
                            "trust_tx_oi": trust_oi,
                            "strategy_signal": signal,
                        }

                    import requests


                    def get_taifex_pcr_and_vix(query_date_str):
                        """從台灣期交所公開網址抓取當日 TXO Put/Call Ratio 與 VIXTWN 收盤指數

                        query_date_str: 'YYYY/MM/DD' 格式
                        """

                        pcr_url = "https://www.taifex.com.tw/cht/3/pcRatioDown"
                        pcr_payload = {
                            "queryStartDate": query_date_str,
                            "queryEndDate": query_date_str,
                        }
                        pcr_oi_ratio = None
                        try:
                            resp_pcr = requests.post(
                                pcr_url, data=pcr_payload, headers=headers, timeout=15
                            )
                            resp_pcr.encoding = "big5"
                            df_pcr = pd.read_csv(io.StringIO(resp_pcr.text), index_col=False)
                            df_pcr.columns = [c.strip() for c in df_pcr.columns]
                            if not df_pcr.empty:
                                # 欄位名稱通常為 '買賣權未平倉量比率%'
                                pcr_oi_ratio = float(df_pcr["買賣權未平倉量比率%"].iloc[0])
                        except Exception as e:
                            print(f"PCR 抓取失敗: {e}")

                        # 2. 抓取臺指選擇權波動率指數 (VIXTWN)
                        # 期交所每日提供當月 VIX 歷史/當日 CSV 下載

                        vix_url = f"https://www.taifex.com.tw/file/taifex/Dailydownload/vix/log2data/{yyyymm}new.txt"
                        vix_close = None
                        try:
                            resp_vix = requests.get(vix_url, headers=headers, timeout=15)
                            resp_vix.encoding = "big5"
                            target_date_compact = query_date_str.replace("/", "")
                            for line in resp_vix.text.splitlines():

                                    len(parts) >= 3
                                    and parts[0] == target_date_compact
                                    and parts[1].startswith("1345")

                                    break
                        except Exception as e:

import datetime as dt
                    import json
                    import os

                    import numpy as np
                    import pandas as pd
                    import requests
                    from scipy.stats import norm
                    import yfinance as yf


                    def calc_bs_gamma(S, K, T, r, sigma):
                        T = np.maximum(T, 1e-5)
                        sigma = np.maximum(sigma, 1e-4)
                        d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
                        return norm.pdf(d1) / (S * sigma * np.sqrt(T))


                    def get_us_ticker_gex(symbol, max_dte=35, r=0.042):
                        ticker = yf.Ticker(symbol)
                        spot = float(ticker.history(period="1d")["Close"].iloc[-1])
                        today = dt.datetime.now(dt.timezone.utc).date()

                        records = []
                        for exp_str in ticker.options:
                            dte = (dt.datetime.strptime(exp_str, "%Y-%m-%d").date() - today).days
                            if dte < 0 or dte > max_dte:
                                continue
                            T = max(dte, 0.5) / 365.0
                            chain = ticker.option_chain(exp_str)
                            for opt_type, df_opt in [("call", chain.calls), ("put", chain.puts)]:
                                sub = df_opt[["strike", "openInterest", "impliedVolatility"]].copy()
                                sub["type"] = opt_type
                                sub["T"] = T
                                records.append(sub)

                        df = pd.concat(records, ignore_index=True).dropna()
                        df = df[(df["openInterest"] > 0) & (df["impliedVolatility"] > 0.01)]
                        df["gamma"] = calc_bs_gamma(
                            spot,
                            df["strike"].values,
                            df["T"].values,
                            r,
                            df["impliedVolatility"].values,
                        )
                        sign = np.where(df["type"] == "call", 1.0, -1.0)
                        df["gex"] = (
                            sign * df["gamma"] * df["openInterest"] * 100 * (spot**2) * 0.01
                        ) / 1e9

                        by_strike = (
                            df.groupby("strike")
                            .agg(
                                net_gex=("gex", "sum"),
                                call_gex=("gex", lambda x: x[x > 0].sum()),
                                put_gex=("gex", lambda x: x[x < 0].sum()),
                            )
                            .reset_index()
                        )

                        by_strike = (
                            by_strike[
                                (by_strike["strike"] >= spot * 0.90)
                                & (by_strike["strike"] <= spot * 1.10)
                            ]
                            .sort_values("strike")
                            .reset_index(drop=True)
                        )

                        call_wall = float(by_strike.loc[by_strike["call_gex"].idxmax(), "strike"])
                        put_wall = float(by_strike.loc[by_strike["put_gex"].idxmin(), "strike"])

                        by_strike["smooth"] = (
                            by_strike["net_gex"].rolling(3, min_periods=1, center=True).mean()
                        )
                        gamma_flip = None
                        for i in range(len(by_strike) - 1):
                            g1, g2 = by_strike.loc[i, "smooth"], by_strike.loc[i + 1, "smooth"]
                            k1, k2 = by_strike.loc[i, "strike"], by_strike.loc[i + 1, "strike"]
                            if g1 < 0 and g2 >= 0:
                                gamma_flip = round(float(k1 + (0 - g1) * (k2 - k1) / (g2 - g1)), 2)
                                if abs(gamma_flip - spot) < spot * 0.06:
                                    break

                        net_total = round(float(by_strike["net_gex"].sum()), 2)
                        regime = (
                            "正 Gamma"
                            if (gamma_flip is None or spot > gamma_flip)
                            else "負 Gamma (高波動警戒)"
                        )
                        return {
                            "symbol": symbol,
                            "spot": round(spot, 2),
                            "call_wall": call_wall,
                            "put_wall": put_wall,
                            "gamma_flip": gamma_flip or "N/A",
                            "net_gex_b": net_total,
                            "regime": regime,
                        }
